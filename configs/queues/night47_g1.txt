# GPU1: latency + FP8 for the DPO A model (alone on the GPU, concurrency 1 = per-question latency). dpo_a_bf16 / dpo_a_fp8 are symlinks of dpo_a (distinct run keys). Same weights; fp8 = vLLM --quantization fp8.
sh mk-links cd /home/app/phunguyen/law_rag/data/models && ln -sfn dpo_a dpo_a_bf16 && ln -sfn dpo_a dpo_a_fp8
wait ! tmux ls 2>/dev/null | grep -q "lawrag-run-mk-links"
serve /home/app/phunguyen/law_rag/data/models/dpo_a_bf16 --gpu 1 --share 0.45

wait ! tmux ls 2>/dev/null | grep -q "lawrag-run-A5_v3b.*-g1"
